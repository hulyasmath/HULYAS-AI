Quick Start Example:
// Include the library


// Create engine
const engine = new ZeqPhysics.Engine();

// Create quantum particle
const particle = engine.createQuantumParticle({
position: new ZeqPhysics.Vector(100, 100),
velocity: new ZeqPhysics.Vector(2, 0),
mass: 1,
radius: 10
});

// Create classical body
const body = engine.createClassicalBody({
position: new ZeqPhysics.Vector(200, 200),
mass: 20,
radius: 15
});

// Add to engine
engine.addBody(particle);
engine.addBody(body);

// Update loop
function animate() {
engine.update();
requestAnimationFrame(animate);
}
animate();
